   602→model Subnet {
   603→  id          String   @id @default(cuid())
   604→  tenantId    String
   605→  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   606→  name        String
   607→  network     String   // e.g. 192.168.1.0
   608→  cidr        Int      // e.g. 24 (for /24)
   609→  cidrNotation String // computed "192.168.1.0/24" for quick lookups
   610→  gateway     String?
   611→  dnsPrimary  String?
   612→  dnsSecondary String?
   613→  vlanId      Int?
   614→  type        String   @default("data") // data | voice | management | guest | pppoe
   615→  status      String   @default("active") // active | reserved | exhausted | archived
   616→  description String?
   617→  totalAddresses Int   @default(0) // 2^(32-cidr)
   618→  usableAddresses Int @default(0) // total - 2 (network + broadcast)
   619→  allocatedCount Int   @default(0)
   620→  createdAt   DateTime @default(now())
   621→  updatedAt   DateTime @updatedAt
   622→
   623→  ipAddresses IpAddress[]
   624→  dhcpLeases  DhcpLease[]
   625→
   626→  @@unique([tenantId, cidrNotation])
   627→  @@index([tenantId, status])
   628→  @@index([tenantId, type])
   629→}

   631→model IpAddress {
   632→  id          String   @id @default(cuid())
   633→  tenantId    String
   634→  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   635→  subnetId    String
   636→  subnet      Subnet   @relation(fields: [subnetId], references: [id], onDelete: Cascade)
   637→  ipAddress    String   @unique
   638→  status      String   @default("available") // available | allocated | reserved | excluded
   639→  assignedTo  String?  // subscriberId or hostname or device name
   640→  assignedType String? // subscriber | nas | device | static | dhcp
   641→  macAddress  String?
   642→  hostname    String?
   643→  notes       String?
   644→  allocatedAt DateTime?
   645→  allocatedBy String?  // userId or "system"
   646→  createdAt   DateTime @default(now())
   647→  updatedAt   DateTime @updatedAt
   648→
   649→  @@index([subnetId, status])
   650→  @@index([tenantId, status])
   651→  @@index([assignedTo])
   652→}

   654→model DhcpLease {
   655→  id          String   @id @default(cuid())
   656→  tenantId    String
   657→  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   658→  subnetId    String?
   659→  subnet      Subnet?  @relation(fields: [subnetId], references: [id], onDelete: SetNull)
   660→  ipAddress    String   @unique
   661→  macAddress  String
   662→  hostname    String?
   663→  clientId    String?  // DHCP Client Identifier
   664→  leaseStart  DateTime @default(now())
   665→  leaseEnd    DateTime
   666→  leaseTime   Int      // seconds
   667→  state       String   @default("active") // active | expired | released | declined
   668→  nasId       String?
   669→  createdAt   DateTime @default(now())
   670→  updatedAt   DateTime @updatedAt
   671→
   672→  @@index([tenantId, state])
   673→  @@index([subnetId])
   674→  @@index([macAddress])
   675→  @@index([leaseEnd])

   678→model DnsZone {
   679→  id          String   @id @default(cuid())
   680→  tenantId    String
   681→  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   682→  name        String   @unique // e.g. example.com
   683→  type        String   @default("forward") // forward | reverse
   684→  soaSerial   Int      @default(1)
   685→  soaRefresh  Int      @default(3600)
   686→  soaRetry    Int      @default(900)
   687→  soaExpire   Int      @default(604800)
   688→  soaMinimum  Int      @default(86400)
   689→  primaryNs   String?  // primary nameserver
   690→  adminEmail  String?  // responsible person
   691→  status      String   @default("active") // active | disabled
   692→  description String?
   693→  createdAt   DateTime @default(now())
   694→  updatedAt   DateTime @updatedAt
   695→
   696→  records     DnsRecord[]
   697→
   698→  @@index([tenantId, status])
   699→}

   701→model DnsRecord {
   702→  id          String   @id @default(cuid())
   703→  tenantId    String
   704→  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   705→  zoneId      String
   706→  zone        DnsZone  @relation(fields: [zoneId], references: [id], onDelete: Cascade)
   707→  name        String   // record name (e.g. "www" or "@" for apex)
   708→  type        String   // A | AAAA | CNAME | MX | TXT | NS | SRV | PTR
   709→  value       String   // record value
   710→  ttl         Int      @default(3600)
   711→  priority    Int?     // for MX records
   712→  weight      Int?     // for SRV records
   713→  port        Int?     // for SRV records
   714→  status      String   @default("active")
   715→  createdAt   DateTime @default(now())
   716→  updatedAt   DateTime @updatedAt
   717→
   718→  @@index([zoneId, type])
   719→  @@index([tenantId, name])
   720→}

   744→  description String?
   745→  lastUpdated DateTime?
   746→  createdAt   DateTime @default(now())

   749→  @@index([tenantId, enabled])
