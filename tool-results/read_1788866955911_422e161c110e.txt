   116→   116→async function fetchPromotions(params: {
   117→   117→  page: number;
   118→   118→  pageSize: number;
   119→   119→  search: string;
   120→   120→  status: string;
   121→   121→  type: string;
   122→   122→}): Promise<{ data: PromotionItem[]; total: number }> {
   123→   123→  const url = new URL("/api/v1/promotions", window.location.origin);
   124→   124→  url.searchParams.set("page", String(params.page));
   125→   125→  url.searchParams.set("pageSize", String(params.pageSize));
   126→   126→  if (params.search) url.searchParams.set("search", params.search);
   127→   127→  if (params.status && params.status !== "all") url.searchParams.set("status", params.status);
   128→   128→  if (params.type && params.type !== "all") url.searchParams.set("type", params.type);
   129→   129→  const res = await fetch(url, { cache: "no-store" });
   130→   130→  if (!res.ok) throw new Error("Failed to fetch promotions");
   131→   131→  const json = await res.json();
   132→   132→  return { data: json.data, total: json.meta.total };
   133→   133→}

   135→   135→async function savePromotion(
   136→   136→  values: PromotionFormValues & { id?: string }
   137→   137→): Promise<void> {
   138→   138→  const isEdit = !!values.id;
   139→   139→  const res = await fetch(
   140→   140→    isEdit ? `/api/v1/promotions/${values.id}` : "/api/v1/promotions",
   141→   141→    {
   142→   142→      method: isEdit ? "PATCH" : "POST",
   143→   143→      headers: { "Content-Type": "application/json" },
   144→   144→      body: JSON.stringify(values),
   145→   145→    }
   146→   146→  );
   147→   147→  if (!res.ok) {
   148→   148→    const err = await res.json().catch(() => ({}));
   149→   149→    throw new Error(err?.error?.message ?? "Failed to save promotion");
   150→   150→  }
   151→   151→}

   153→   153→async function deletePromotion(id: string): Promise<void> {
   154→   154→  const res = await fetch(`/api/v1/promotions/${id}`, { method: "DELETE" });
   155→   155→  if (!res.ok) {
   156→   156→    const err = await res.json().catch(() => ({}));
   157→   157→    throw new Error(err?.error?.message ?? "Failed to delete promotion");
   158→   158→  }
   159→   159→}

   499→   499→
   500→   500→      {/* Create dialog */}
   501→   501→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>

   504→   504→            <DialogTitle className="flex items-center gap-2">
   505→   505→              <Tag className="h-5 w-5 text-brand" /> New Promotion
   506→   506→            </DialogTitle>
   507→   507→            <DialogDescription>
   508→   508→              Discount codes apply at checkout. Codes are auto-uppercased.
   509→   509→            </DialogDescription>
   510→   510→          </DialogHeader>
   511→   511→          <PromotionForm
   512→   512→            isSaving={saveMutation.isPending}

   514→   514→          />
   515→   515→        </DialogContent>
   516→   516→      </Dialog>
   517→   517→
   518→   518→      {/* Edit dialog */}

   537→   537→              key={editTarget.id}
   538→   538→              initial={editTarget}
   539→   539→              isSaving={saveMutation.isPending}
   540→   540→              onSave={(values) =>
   541→   541→                saveMutation.mutate({ ...values, id: editTarget.id })

   543→   543→            />

   550→   550→        open={!!deleteTarget}

   556→   556→          <AlertDialogHeader>

   562→   562→          </AlertDialogHeader>

   573→   573→        </AlertDialogContent>

   575→   575→    </>
