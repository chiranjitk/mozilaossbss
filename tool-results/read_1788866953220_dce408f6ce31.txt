   116→async function fetchPromotions(params: {
   117→  page: number;
   118→  pageSize: number;
   119→  search: string;
   120→  status: string;
   121→  type: string;
   122→}): Promise<{ data: PromotionItem[]; total: number }> {
   123→  const url = new URL("/api/v1/promotions", window.location.origin);
   124→  url.searchParams.set("page", String(params.page));
   125→  url.searchParams.set("pageSize", String(params.pageSize));
   126→  if (params.search) url.searchParams.set("search", params.search);
   127→  if (params.status && params.status !== "all") url.searchParams.set("status", params.status);
   128→  if (params.type && params.type !== "all") url.searchParams.set("type", params.type);
   129→  const res = await fetch(url, { cache: "no-store" });
   130→  if (!res.ok) throw new Error("Failed to fetch promotions");
   131→  const json = await res.json();
   132→  return { data: json.data, total: json.meta.total };
   133→}

   135→async function savePromotion(
   136→  values: PromotionFormValues & { id?: string }
   137→): Promise<void> {
   138→  const isEdit = !!values.id;
   139→  const res = await fetch(
   140→    isEdit ? `/api/v1/promotions/${values.id}` : "/api/v1/promotions",
   141→    {
   142→      method: isEdit ? "PATCH" : "POST",
   143→      headers: { "Content-Type": "application/json" },
   144→      body: JSON.stringify(values),
   145→    }
   146→  );
   147→  if (!res.ok) {
   148→    const err = await res.json().catch(() => ({}));
   149→    throw new Error(err?.error?.message ?? "Failed to save promotion");
   150→  }
   151→}

   153→async function deletePromotion(id: string): Promise<void> {
   154→  const res = await fetch(`/api/v1/promotions/${id}`, { method: "DELETE" });
   155→  if (!res.ok) {
   156→    const err = await res.json().catch(() => ({}));
   157→    throw new Error(err?.error?.message ?? "Failed to delete promotion");
   158→  }
   159→}

   583→function PromotionForm({
   584→  initial,
   585→  isSaving,
   586→  onSave,
   587→}: {
   588→  initial?: PromotionItem;
   589→  isSaving: boolean;
   590→  onSave: (values: PromotionFormValues) => void;
   591→}) {
   592→  const toLocalInput = (date: string | Date) => {
   593→    const d = typeof date === "string" ? new Date(date) : date;
   594→    return d.toISOString().slice(0, 16);
   595→  };
   596→
   597→  const [form, setForm] = useState<{
   598→    name: string;
   599→    code: string;
   600→    description: string;
   601→    type: PromotionFormValues["type"];
   602→    value: string;
   603→    maxUses: string;
   604→    validFrom: string;
   605→    validUntil: string;
   606→    status: PromotionFormValues["status"];
   607→    applicablePlans: string;
   608→  }>({
   609→    name: initial?.name ?? "",
   610→    code: initial?.code ?? "",
   611→    description: initial?.description ?? "",
   612→    type: (initial?.type as PromotionFormValues["type"]) ?? "percentage",
   613→    value: initial?.value?.toString() ?? "10",
